\section{Data Tensors}
For episode $e$ and frame index $t$:
\begin{align}
x_{e,t} &\in \{0,\ldots,255\}^{72 \times 128 \times 3}, \\
a_{e,t} &\in \mathbb{R}^{3}, \\
s_{e,t} &\in \mathbb{R}^{16 \times 4}.
\end{align}

Action vector:
\begin{align}
a_{e,t} = [f_x, f_y, \text{trigger}],
\end{align}
where trigger is $1.0$ on shot-release frames, otherwise $0.0$.

Simulation state per ball stores:
\begin{align}
[\text{pos}_x,\text{pos}_y,\text{vel}_x,\text{vel}_y].
\end{align}

\section{VAE}
\subsection{Encoder/Decoder Geometry}
The encoder downsamples spatially by factor $8$ (three stride-$2$ stages):
\begin{align}
72 \times 128 \rightarrow 36 \times 64 \rightarrow 18 \times 32 \rightarrow 9 \times 16.
\end{align}

Approximate posterior:
\begin{align}
q_\phi(z\mid x) = \mathcal{N}\!\left(\mu_\phi(x), \operatorname{diag}(\sigma_\phi^2(x))\right),\quad
\sigma_\phi(x)=\exp\!\left(\tfrac12 \log \sigma_\phi^2(x)\right),
\end{align}
with reparameterization:
\begin{align}
z = \mu_\phi(x) + \sigma_\phi(x)\odot \epsilon,\quad \epsilon\sim\mathcal{N}(0,I).
\end{align}

Decoder:
\begin{align}
\hat{x} = g_\theta(z), \quad z\in\mathbb{R}^{4\times 9\times16}.
\end{align}

\subsection{VAE Loss}
The implemented loss is:
\begin{align}
\mathcal{L}_{\text{VAE}}
=
\lambda_{L1}\,\|x-\hat{x}\|_1
 + \lambda_{\text{LPIPS}}\,\mathcal{L}_{\text{LPIPS}}(x,\hat{x})
 + \lambda_{\text{KL}}\,\beta(n)\,\mathcal{L}_{\text{KL}},
\end{align}
where
\begin{align}
\mathcal{L}_{\text{KL}}
&= -\frac12\,\mathbb{E}\left[1+\log\sigma^2-\mu^2-\sigma^2\right], \\
\beta(n)
&= \beta_{\text{start}}
+ \min\!\left(1,\frac{n}{N_{\text{warmup}}}\right)\left(\beta_{\text{end}}-\beta_{\text{start}}\right).
\end{align}

Current values:
\begin{align}
\lambda_{L1}=1.0,\;
\lambda_{\text{LPIPS}}=0.1,\;
\lambda_{\text{KL}}=1.0,\;
\beta_{\text{start}}=10^{-6},\;
\beta_{\text{end}}=5\times10^{-4},\;
N_{\text{warmup}}=300{,}000\ \text{frames}.
\end{align}

\section{Latent Training Samples}
After VAE export, latent shards provide:
\begin{align}
z_{e,t}\in \mathbb{R}^{4\times 9\times 16}, \quad
a_{e,t}\in\mathbb{R}^{3}.
\end{align}

Baseline training sample at time $t$:
\begin{align}
\text{context}_t &= [z_{t-L+1},\ldots,z_t]\in\mathbb{R}^{L\times C\times H_z\times W_z}, \\
\text{action}_t &= a_t\in\mathbb{R}^3, \\
\text{target}_t &= z_{t+1}\in\mathbb{R}^{C\times H_z\times W_z}.
\end{align}

Diffusion-forcing sample additionally uses:
\begin{align}
\text{actions\_seq}_t=[a_t,\ldots,a_{t+K-1}],\quad
\text{targets\_seq}_t=[z_{t+1},\ldots,z_{t+K}].
\end{align}

\section{Diffusion in Latent Space}
Define linear noise schedule:
\begin{align}
\beta_1,\ldots,\beta_T \text{ linearly spaced from } 10^{-4} \text{ to } 0.02,\quad
\alpha_t = 1-\beta_t,\quad
\bar{\alpha}_t = \prod_{i=1}^t \alpha_i.
\end{align}

Forward corruption of clean latent $x_0$:
\begin{align}
x_t = \sqrt{\bar{\alpha}_t}\,x_0 + \sqrt{1-\bar{\alpha}_t}\,\epsilon,\quad
\epsilon\sim\mathcal{N}(0,I).
\end{align}

Model predicts $\hat{\epsilon}_\theta$ and baseline objective is:
\begin{align}
\mathcal{L}_{\epsilon}
=
\mathbb{E}_{t,\epsilon}\left[\|\hat{\epsilon}_\theta - \epsilon\|_2^2\right].
\end{align}

Estimated clean latent from $(x_t,\hat{\epsilon})$:
\begin{align}
\hat{x}_0 = \frac{x_t-\sqrt{1-\bar{\alpha}_t}\,\hat{\epsilon}}{\sqrt{\bar{\alpha}_t}}.
\end{align}

\section{Action-Conditioned DiT}
Flatten each latent frame:
\begin{align}
\mathrm{vec}(z)\in\mathbb{R}^{d_z},\quad d_z=576.
\end{align}

Tokenization:
\begin{align}
h_i &= W_c\,\mathrm{vec}(z_{t-L+i}) \in \mathbb{R}^{d_{\text{model}}},\quad i=1,\dots,L,\\
h_{\text{tgt}} &= W_t\,\mathrm{vec}(x_t) + E_a(a_t) + E_t(t)\in\mathbb{R}^{d_{\text{model}}}.
\end{align}

Transformer input:
\begin{align}
H = [h_1,\ldots,h_L,h_{\text{tgt}}] + P,
\end{align}
then
\begin{align}
\hat{\epsilon} = \mathrm{unvec}\!\left(W_o\,\mathrm{Transformer}(H)_{\text{last}}\right)
\in\mathbb{R}^{4\times9\times16}.
\end{align}

\section{Diffusion Forcing (DF)}
With rollout length $K$, for each rollout step $k=1,\dots,K$:
\begin{align}
\mathcal{L}_k = \|\hat{\epsilon}_k-\epsilon_k\|_2^2.
\end{align}

Teacher forcing probability as a function of processed samples $n$:
\begin{align}
p_{\text{tf}}(n)
= p_0 + \min\!\left(1,\frac{n}{N_d}\right)(p_1-p_0),
\end{align}
with current values:
\begin{align}
p_0=0.98,\quad p_1=0.05,\quad N_d=420{,}000{,}000.
\end{align}

Context update between rollout steps:
\begin{align}
\tilde{z}_{k+1}
= m_k \odot z_{k+1}^{\text{gt}} + (1-m_k)\odot \hat{x}_{0,k},
\quad
m_k\sim \operatorname{Bernoulli}(p_{\text{tf}}).
\end{align}

Then
\begin{align}
\text{context} \leftarrow [\text{context}_{2:L}, \tilde{z}_{k+1}].
\end{align}

Final DF objective:
\begin{align}
\mathcal{L}_{\text{DF}}=\frac{1}{K}\sum_{k=1}^{K}\mathcal{L}_k.
\end{align}

Implementation detail currently enabled:
\begin{align}
\hat{x}_{0,k}\ \text{is detached before context insertion.}
\end{align}

\section{Distributed Batch Math}
If per-rank batch is $B_{\text{local}}$ and world size is $R$:
\begin{align}
B_{\text{global}} = B_{\text{local}} \cdot R.
\end{align}
Current run uses $B_{\text{local}}=512$, $R=2$, so $B_{\text{global}}=1024$.

\section{What We Are Using}
The current world-model training mode in this repository is:

\textbf{DiT + Diffusion Forcing}
\end{align}
with latent-space diffusion loss, action conditioning, rollout training ($K=4$), and decayed teacher forcing.
